

SLOW = '"learning_rate":0.03,"feature_fraction":0.55,"num_leaves":95,"lambda_l2":30'
BIG = '"learning_rate":0.015,"num_leaves":127,"feature_fraction":0.5,"lambda_l2":30,"min_data_in_leaf":200,"max_bin":255,"force_col_wise":true'
LP = '"max_bin":63,"force_col_wise":true'
LR = 'feat_lr2,feat_tt,feat_pp,feat_sqC,feat_dw'
DROP = "'^sq_(pdot|pcos|mdot|mcos)$'"
def pipeline(W, RAW, T, PY, OUT, cut):
    """ordered (stage, shell command or None for setup, subdir of W)"""
    FOLDS = 'f3 f4 f5 f6 eval' if cut == 27 else 'f1 f2 f3 f4 val25'
    V2 = 'lite' if cut == 27 else 'c25'
    return [('setup', None, ''), ('prep_v3', f'{PY} prep.py {RAW}', 'v3'), ('prep_v2', f'{PY} prep.py', ''),
            ('feat_v3', f'{PY} v3fold.py {FOLDS}', ''), ('feat_v2', f'{PY} build_folds.py {V2}', ''),
            ('feat_x', f'{PY} extra.py {FOLDS}', ''), ('feat_lr', f'LR_KS=4,6,8,10,12,16 LR_CON=1 {PY} lrfeat2.py {FOLDS}', ''),
            ('feat_tt', f'TT_THREADS={T} {PY} ttfeat.py {FOLDS}', ''), ('feat_pp', f'{PY} ppmi.py {FOLDS}', ''),
            ('feat_sq', f'V12_THREADS={T} {PY} seqrun.py {FOLDS}', ''), ('feat_dw', f'{PY} dw/mkdw.py {FOLDS}', ''),
            ('dwbig', f"{PY} ens.py {cut} dwbig --seeds 0 --rounds 1000 --save 1000 --threads {T} --lr {LR} --nfolds 4 --drop {DROP} --params '{{{BIG}}}'", ''),
            ('dw_s', f"{PY} ens.py {cut} dw_s --seeds 0 --rounds 500 --save 500 --threads {T} --lr {LR} --nfolds 4 --negfrac 0.5 --drop {DROP} --params '{{{SLOW},{LP}}}'", ''),
            *[(f'mlp{s}', f'MLP_THREADS={T} {PY} mlp12.py {cut} dwmlp --ep 4 --seed {s} --lr {LR}', '') for s in range(3)],
            ('blend', f'{PY} blend12.py {cut}', ''),
            *([('write', f'PRED={W}/eval_pred.parquet {PY} write_sub.py {OUT}', ''), ('validate', f'{PY} validate.py {OUT}', '')] if cut == 27 else [])]
def setup(W, RAW):
    os.makedirs(os.path.join(W, 'v3'), exist_ok=True); os.makedirs(os.path.join(W, 'ens'), exist_ok=True)
    for k, s in SRC.items():
        os.makedirs(os.path.dirname(os.path.join(W, k)), exist_ok=True)
        open(os.path.join(W, k), 'w').write(s.replace('@@WORK@@', W).replace('@@RAW@@', RAW))
    for f in ('train.csv', 'evaluation.csv'):          # v2 prep / writer read the raw CSVs from WORK
        dst = os.path.join(W, f)
        if not os.path.exists(dst): os.symlink(os.path.join(RAW, f), dst)
def thread_env(T):
    return dict(os.environ, POLARS_MAX_THREADS=T, OMP_NUM_THREADS=T, OPENBLAS_NUM_THREADS=T, MKL_NUM_THREADS=T, NUMEXPR_NUM_THREADS=T,
                TT_THREADS=T, SQ_THREADS=T, RAYON_NUM_THREADS=T, PYTHONUNBUFFERED='1')
def main():
    ap = argparse.ArgumentParser(); ap.add_argument('--raw', required=True); ap.add_argument('--work', default='./v12_work')
    ap.add_argument('--threads', type=int, default=8); ap.add_argument('--out', default=None); ap.add_argument('--from', dest='frm', default='setup'); ap.add_argument('--to', default=None)
    ap.add_argument('--cut', type=int, default=27, choices=(25, 27))
    a = ap.parse_args()
    W = os.path.abspath(a.work); RAW = os.path.abspath(a.raw); T = str(a.threads)
    OUT = os.path.abspath(a.out or os.path.join(W, 'submission_v12.csv'))
    os.makedirs(os.path.dirname(OUT), exist_ok=True)
    env = thread_env(T); PY = sys.executable
    P = pipeline(W, RAW, T, PY, OUT, a.cut); names = [p[0] for p in P]; to = a.to or names[-1]
    tf = os.path.join(W, 'timings.json'); tim = json.load(open(tf)) if os.path.exists(tf) and a.frm != 'setup' else {}
    t_all = time.time()
    for st, cmd, sub in P[names.index(a.frm):names.index(to) + 1]:
        t0 = time.time()
        if cmd is None: setup(W, RAW)
        else: print('>>', cmd, flush=True); subprocess.run(cmd, shell=True, check=True, cwd=os.path.join(W, sub), env=env)
        tim[st] = round(time.time() - t0, 1); print(f'[stage {st}] {tim[st]:.0f}s', flush=True)
        os.makedirs(W, exist_ok=True); json.dump(tim, open(tf, 'w'), indent=1)
    print('TOTAL this run %.1f min; all stages %.1f min' % ((time.time() - t_all) / 60, sum(tim.values()) / 60))
    if a.cut == 27: print('submission:', OUT)
if __name__ == '__main__':
    main()
